cmake_minimum_required(VERSION 3.24)
project(host_optimization_flags LANGUAGES CXX)

include("${OPTIMIZATION_MODULE}")
set(CMAKE_CXX_COMPILER_ID AppleClang)
add_library(external INTERFACE)
# Check the real helper's target properties without compiling emulator dependencies.
add_library(optimized_target STATIC dummy.cpp)
add_library(control_target STATIC dummy.cpp)
set(ENABLE_HOST_OPTIMIZATIONS OFF)
shadps4_optimize_host_target(control_target)
set(ENABLE_HOST_OPTIMIZATIONS ON)
shadps4_optimize_host_target(optimized_target)

get_target_property(compile_options optimized_target COMPILE_OPTIONS)
get_target_property(link_options optimized_target LINK_OPTIONS)
get_target_property(control_options control_target COMPILE_OPTIONS)
get_target_property(external_options external INTERFACE_COMPILE_OPTIONS)
if(NOT compile_options STREQUAL "$<$<COMPILE_LANGUAGE:CXX>:-O3>;$<$<COMPILE_LANGUAGE:CXX>:-flto=thin>")
    message(FATAL_ERROR "Unexpected optimized compile flags: ${compile_options}")
endif()
if(NOT link_options STREQUAL "-flto=thin")
    message(FATAL_ERROR "Missing ThinLTO link flag")
endif()
if(control_options OR external_options)
    message(FATAL_ERROR "Host flags leaked into disabled or external targets")
endif()
set(CMAKE_CXX_COMPILER_ID GNU)
add_library(non_clang STATIC dummy.cpp)
shadps4_optimize_host_target(non_clang)
get_target_property(non_clang_options non_clang COMPILE_OPTIONS)
if(non_clang_options)
    message(FATAL_ERROR "Clang flags applied to another compiler")
endif()
# Profile-guided builds: GENERATE instruments, USE needs a profile file, OFF adds nothing.
set(CMAKE_CXX_COMPILER_ID AppleClang)
set(ENABLE_HOST_OPTIMIZATIONS OFF)
add_library(pgo_generate STATIC dummy.cpp)
set(SHADPS4_PGO GENERATE)
shadps4_optimize_host_target(pgo_generate)
get_target_property(generate_options pgo_generate COMPILE_OPTIONS)
get_target_property(generate_links pgo_generate LINK_OPTIONS)
if(NOT generate_options STREQUAL "$<$<COMPILE_LANGUAGE:CXX>:-fprofile-generate>" OR
   NOT generate_links STREQUAL "-fprofile-generate")
    message(FATAL_ERROR "Unexpected profile generate flags: ${generate_options} ${generate_links}")
endif()
add_library(pgo_use STATIC dummy.cpp)
set(SHADPS4_PGO USE)
set(SHADPS4_PGO_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/dummy.cpp")
shadps4_optimize_host_target(pgo_use)
get_target_property(use_links pgo_use LINK_OPTIONS)
if(NOT use_links STREQUAL "-fprofile-use=${SHADPS4_PGO_PROFILE}")
    message(FATAL_ERROR "Unexpected profile use link flags: ${use_links}")
endif()
add_library(pgo_off STATIC dummy.cpp)
set(SHADPS4_PGO OFF)
shadps4_optimize_host_target(pgo_off)
get_target_property(off_options pgo_off COMPILE_OPTIONS)
if(off_options)
    message(FATAL_ERROR "Profile flags applied with SHADPS4_PGO=OFF: ${off_options}")
endif()
set_target_properties(optimized_target control_target non_clang pgo_generate pgo_use pgo_off
                      PROPERTIES LINKER_LANGUAGE CXX)
